
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

#define MAX 50

// Structure to store student details
struct Student
{
    int roll;
    char name[MAX];
    char department[MAX];
    int year;
    float marks;
    char grade[5];
};

typedef struct Student Student;

// Global variables
Student *students;
int count = 0;
int capacity = 2;

// Function declarations
void loadRecords();
void saveRecords();
void addStudent();
void displayStudents();
void searchStudent();
void updateStudent();
void deleteStudent();
void countStudents();
void filterStudents();
void assignGrade(Student *s);
void showMenu();


// Assign grade based on marks
void assignGrade(Student *s)
{
    if (s->marks >= 90)
        strcpy(s->grade, "A+");
    else if (s->marks >= 80)
        strcpy(s->grade, "A");
    else if (s->marks >= 70)
        strcpy(s->grade, "B");
    else if (s->marks >= 60)
        strcpy(s->grade, "C");
    else if (s->marks >= 50)
        strcpy(s->grade, "D");
    else
        strcpy(s->grade, "F");
}


// Load records from binary file
void loadRecords()
{
    FILE *fp;

    fp = fopen("students.dat", "rb");

    if (fp == NULL)
        return;

    while (1)
    {
        Student temp;

        if (fread(&temp, sizeof(Student), 1, fp) != 1)
            break;

        if (count == capacity)
        {
            Student *ptr = realloc(
                students,
                capacity * 2 * sizeof(Student)
            );

            if (ptr == NULL)
            {
                printf("Memory allocation failed!\n");
                fclose(fp);
                return;
            }

            students = ptr;
            capacity = capacity * 2;
        }

        students[count] = temp;
        count++;
    }

    fclose(fp);
}


// Save records into binary and text files
void saveRecords()
{
    FILE *fp;
    FILE *tf;

    fp = fopen("students.dat", "wb");

    if (fp == NULL)
    {
        printf("Cannot open binary file!\n");
        return;
    }

    if (fwrite(students, sizeof(Student), count, fp) != count)
        printf("Binary file writing error!\n");

    fclose(fp);

    tf = fopen("students.txt", "w");

    if (tf == NULL)
    {
        printf("Cannot open text file!\n");
        return;
    }

    for (int i = 0; i < count; i++)
    {
        fprintf(tf, "%d|%s|%s|%d|%.2f|%s\n",
                students[i].roll,
                students[i].name,
                students[i].department,
                students[i].year,
                students[i].marks,
                students[i].grade);
    }

    fclose(tf);
}


// Add a new student
void addStudent()
{
    Student *newStudent;

    newStudent = malloc(sizeof(Student));

    if (newStudent == NULL)
    {
        printf("Memory allocation failed!\n");
        return;
    }

    printf("\nEnter Roll Number: ");
    scanf("%d", &newStudent->roll);

    // Check duplicate roll number
    for (int i = 0; i < count; i++)
    {
        if (students[i].roll == newStudent->roll)
        {
            printf("Roll Number already exists!\n");
            free(newStudent);
            return;
        }
    }

    printf("Enter Student Name (one word): ");
    scanf("%49s", newStudent->name);

    printf("Enter Department (one word): ");
    scanf("%49s", newStudent->department);

    printf("Enter Year: ");
    scanf("%d", &newStudent->year);

    printf("Enter Marks: ");
    scanf("%f", &newStudent->marks);

    if (newStudent->year < 1 || newStudent->year > 6 ||
        newStudent->marks < 0 || newStudent->marks > 100)
    {
        printf("Invalid year or marks!\n");
        free(newStudent);
        return;
    }

    assignGrade(newStudent);

    // Increase memory if required
    if (count == capacity)
    {
        Student *ptr = realloc(
            students,
            capacity * 2 * sizeof(Student)
        );

        if (ptr == NULL)
        {
            printf("Memory resizing failed!\n");
            free(newStudent);
            return;
        }

        students = ptr;
        capacity = capacity * 2;
    }

    students[count] = *newStudent;
    count++;

    free(newStudent);

    saveRecords();

    printf("\nStudent added successfully!\n");
    printf("Assigned Grade: %s\n", students[count - 1].grade);
}


// Display all student records using text file
void displayStudents()
{
    FILE *fp;
    Student s;
    int found = 0;

    fp = fopen("students.txt", "r");

    if (fp == NULL)
    {
        printf("\nNo records available!\n");
        return;
    }

    printf("\n----------- STUDENT RECORDS -----------\n");

    printf("%-8s %-15s %-15s %-6s %-8s %-6s\n",
           "Roll", "Name", "Department", "Year", "Marks", "Grade");

    while (fscanf(fp, "%d|%49[^|]|%49[^|]|%d|%f|%4s",
                  &s.roll,
                  s.name,
                  s.department,
                  &s.year,
                  &s.marks,
                  s.grade) == 6)
    {
        printf("%-8d %-15s %-15s %-6d %-8.2f %-6s\n",
               s.roll, s.name, s.department,
               s.year, s.marks, s.grade);

        found = 1;
    }

    if (!found)
        printf("No student records found.\n");

    fclose(fp);
}


// Search student by roll number
void searchStudent()
{
    int roll, found = 0;

    printf("\nEnter Roll Number to Search: ");
    scanf("%d", &roll);

    for (int i = 0; i < count; i++)
    {
        if (students[i].roll == roll)
        {
            printf("\nStudent Found!\n");
            printf("Roll Number : %d\n", students[i].roll);
            printf("Name        : %s\n", students[i].name);
            printf("Department  : %s\n", students[i].department);
            printf("Year        : %d\n", students[i].year);
            printf("Marks       : %.2f\n", students[i].marks);
            printf("Grade       : %s\n", students[i].grade);

            found = 1;
            break;
        }
    }

    if (!found)
        printf("\nStudent not found!\n");
}


// Update student details
void updateStudent()
{
    int roll, found = 0;

    printf("\nEnter Roll Number to Update: ");
    scanf("%d", &roll);

    for (int i = 0; i < count; i++)
    {
        if (students[i].roll == roll)
        {
            printf("Enter New Department: ");
            scanf("%49s", students[i].department);

            printf("Enter New Year: ");
            scanf("%d", &students[i].year);

            printf("Enter New Marks: ");
            scanf("%f", &students[i].marks);

            if (students[i].year < 1 || students[i].year > 6 ||
                students[i].marks < 0 || students[i].marks > 100)
            {
                printf("Invalid year or marks! Update cancelled.\n");
                return;
            }

            assignGrade(&students[i]);

            saveRecords();

            printf("\nStudent record updated successfully!\n");
            found = 1;
            break;
        }
    }

    if (!found)
        printf("\nStudent not found!\n");
}


// Delete student record
void deleteStudent()
{
    int roll, found = 0;

    printf("\nEnter Roll Number to Delete: ");
    scanf("%d", &roll);

    for (int i = 0; i < count; i++)
    {
        if (students[i].roll == roll)
        {
            for (int j = i; j < count - 1; j++)
            {
                students[j] = students[j + 1];
            }

            count--;

            saveRecords();

            printf("\nStudent deleted successfully!\n");
            found = 1;
            break;
        }
    }

    if (!found)
        printf("\nStudent not found!\n");
}


// Count total students
void countStudents()
{
    printf("\nTotal Number of Students: %d\n", count);
}


// Filter students by marks
void filterStudents()
{
    float limit;
    int found = 0;

    printf("\nEnter Minimum Marks: ");
    scanf("%f", &limit);

    printf("\nStudents scoring above %.2f:\n", limit);

    for (int i = 0; i < count; i++)
    {
        if (students[i].marks > limit)
        {
            printf("\nRoll Number : %d", students[i].roll);
            printf("\nName        : %s", students[i].name);
            printf("\nMarks       : %.2f", students[i].marks);
            printf("\nGrade       : %s\n", students[i].grade);

            found = 1;
        }
    }

    if (!found)
        printf("No students found!\n");
}


// Display menu
void showMenu()
{
    static int menuCount = 0;

    menuCount++;

    printf("\n========== STUDENT MANAGEMENT SYSTEM ==========\n");
    printf("1. Add Student Record\n");
    printf("2. Display All Records\n");
    printf("3. Search Student\n");
    printf("4. Update Student Record\n");
    printf("5. Delete Student Record\n");
    printf("6. Count Total Students\n");
    printf("7. Filter Students by Marks\n");
    printf("8. Exit\n");

    printf("\nMenu displayed %d times.\n", menuCount);
}


// Main function
int main()
{
    int choice;

    // Allocate initial memory using calloc
    students = calloc(capacity, sizeof(Student));

    if (students == NULL)
    {
        printf("Memory allocation failed!\n");
        return 1;
    }

    // Load previously saved records
    loadRecords();

    do
    {
        showMenu();

        printf("\nEnter Your Choice: ");
        scanf("%d", &choice);

        switch (choice)
        {
            case 1:
                addStudent();
                break;

            case 2:
                displayStudents();
                break;

            case 3:
                searchStudent();
                break;

            case 4:
                updateStudent();
                break;

            case 5:
                deleteStudent();
                break;

            case 6:
                countStudents();
                break;

            case 7:
                filterStudents();
                break;

            case 8:
                saveRecords();
                free(students);
                printf("\nMemory released successfully!\n");
                printf("Thank you for using the system!\n");
                break;

            default:
                printf("\nInvalid Choice! Try Again.\n");
        }

    } while (choice != 8);

    return 0;
}
